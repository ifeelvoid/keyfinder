# KeyFinder 2.0 Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship KeyFinder 2.0 with live audio input (system audio capture), UI repaint (album art, waveform, prominent key), shared VST/AU plugin, and GitHub issue fixes.

**Architecture:**
- Create `KeyFinderEngine` shared framework (Swift Package) containing KeyDetector, BPMDetector, BeatGridDetector, AudioProcessor
- Standalone app and VST/AU plugin both link against KeyFinderEngine
- Live audio via AVCaptureSession with BackgroundThread analysis (decoupled from audio thread)
- VST plugin uses background queue for analysis; audio thread stays clean
- On first launch, app defaults to Live mode; thereafter remembers last mode

**Tech Stack:** Swift, AVFoundation (AVCaptureSession), Swift Package Manager, JUCE (for VST/AU plugin wrapper), XcodeGen

```swift
import Foundation
import AVFoundation

/// Public entry point for KeyFinder analysis engine
public struct KeyFinderEngine {
    private let audioProcessor: AudioProcessor

    public init() {
        self.audioProcessor = AudioProcessor()
    }

    /// Analyze an audio file and return complete analysis results
    public func analyze(url: URL) async throws -> AnalysisResult {
        return try await audioProcessor.analyzeAudioFile(at: url)
    }

    /// Analyze audio samples directly (for live input or custom sources)
    public func analyze(samples: [Float], sampleRate: Double) async -> AnalysisResult {
        let key = try? await KeyDetector().detectKeyWithChanges(
            audioSamples: samples,
            sampleRate: sampleRate
        )
        let bpm = try? await BPMDetector().detectBPM(
            audioSamples: samples,
            sampleRate: sampleRate
        )
        // Return result with available data
    }

    /// Real-time analysis result for live input
    public struct AnalysisResult {
        public let key: String
        public let camelot: String
        public let bpm: Double
        public let confidence: Double
        public let keyChanges: [(TimeInterval, String, Double)]
        public let beatGrid: BeatGridData?
    }
}
```

```swift
import AVFoundation
import Accelerate

/// Captures system audio via AVCaptureSession (macOS 14+)
/// No third-party drivers needed — uses Core Audio loopback
public class LiveAudioCapture: ObservableObject {
    @Published public var currentKey: String = "--"
    @Published public var currentBPM: String = "--"
    @Published public var isCapturing: Bool = false

    private var captureSession: AVCaptureSession?
    private var audioEngine: AVAudioEngine?
    private let keyDetector = KeyDetector()
    private let bpmDetector = BPMDetector()

    // Ring buffer for audio samples
    private var sampleBuffer: [Float] = []
    private let bufferSize = 48000 * 30 // 30 seconds at 48kHz

    public init() {}

    /// Start capturing from default system audio input
    public func startCapture() {
        // Set up AVCaptureSession for audio
        let session = AVCaptureSession()
        session.sessionPreset = .high

        // Get default audio device
        guard let audioDevice = AVCaptureDevice.default(for: .audio) else {
            print("No audio device found")
            return
        }

        do {
            let input = try AVCaptureDeviceInput(device: audioDevice)
            if session.canAddInput(input) {
                session.addInput(input)
            }

            let output = AVCaptureAudioDataOutput()
            if session.canAddOutput(output) {
                session.addOutput(output)
            }

            output.setSampleBufferDelegate(self, queue: DispatchQueue(label: "audio.capture"))

            session.startRunning()
            self.captureSession = session
            self.isCapturing = true
        } catch {
            print("Failed to set up capture session: \(error)")
        }
    }

    /// Stop capturing
    public func stopCapture() {
        captureSession?.stopRunning()
        captureSession = nil
        isCapturing = false
    }
}

extension LiveAudioCapture: AVCaptureAudioDataOutputSampleBufferDelegate {
    public func captureOutput(_ output: AVCaptureOutput,
                              didOutput sampleBuffer: CMSampleBuffer,
                              from connection: AVCaptureConnection) {
        // Extract audio samples and run analysis
        guard let block = CMSampleBufferGetAudioBufferList(audio: sampleBuffer) else { return }

        // Process samples through analysis engine
        let samples = extractSamples(from: block)

        // Run analysis on background thread
        Task {
            let keyResult = try? await keyDetector.detectKeyWithChanges(
                audioSamples: samples,
                sampleRate: 44100
            )
            let bpm = try? await bpmDetector.detectBPM(
                audioSamples: samples,
                sampleRate: 44100
            )

            await MainActor.run {
                self.currentKey = keyResult?.key.shortName ?? "--"
                self.currentBPM = bpm.map { String(format: "%.1f", $0) } ?? "--"
            }
        }
    }
}
```
